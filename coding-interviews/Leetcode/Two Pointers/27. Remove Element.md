class Solution(object):
    def removeElement(self, nums, val):
        """
        :type nums: List[int]
        :type val: int
        :rtype: int
        """
        k = 0
        for counter in range(len(nums)):
            if nums[counter] != val:
                nums[k] = nums[counter]
                k +=1
        return k
