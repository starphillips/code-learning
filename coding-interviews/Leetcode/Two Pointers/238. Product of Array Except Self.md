class Solution(object):
    def productExceptSelf(self, nums):
        # nums = array
        # answer = array and return 
        # answer[i] = all elemetns of nums except the index its currently on

        # for every index * together 
        # two pointers as you use one pointer to cycle through all elements and one to highlight thats the index you're ignoring. 
        # assign the excluded - e = 0
        # assign the rotator - r = 1
        answer = []
        for exclude in range(len(nums)):
            product = 1
            for j in range(len(nums)):
                if j == exclude:
                    continue
                else:
                    product *= nums[j]
            answer.append(product)
        return answer

        

        """
        :type nums: List[int]
        :rtype: List[int]
        """
