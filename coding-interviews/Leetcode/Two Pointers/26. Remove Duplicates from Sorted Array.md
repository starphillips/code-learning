class Solution(object):
    def removeDuplicates(self, nums):
        k = 0
        counter = 1
        while counter < len(nums):
            if nums[k] != nums[counter]:
                k +=1
                nums[k] = nums[counter]
            counter +=1
        return k+1
        """
        :type nums: List[int]
        :rtype: int
        """
