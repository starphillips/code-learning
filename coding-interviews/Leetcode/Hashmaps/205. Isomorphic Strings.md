class Solution:
    def isIsomorphic(self, s: str, t: str) -> bool:
        # input s, t = strings
        # map to each  
        # output = bool
        s_map = {}
        t_map = {}

        if len(s) != len(t):
            return False
        # zip(s,t) = ('e', 'a'), ('g', 'd'), ('g', 'd')
        for a, b in zip(s, t):
            if a in s_map and s_map[a] != b:
                return False
            if b in t_map and t_map[b] != a:
                return False
            s_map[a] = b
            t_map[b] = a
        return True
