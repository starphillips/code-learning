``` python
class Solution:
    def wordPattern(self, pattern: str, s: str) -> bool:
        pattern_map = {}
        word_map = {}

        words = s.split()

        if len(words) != len(pattern):
            return False

        for a, b in zip(pattern, words):
            if a in pattern_map and pattern_map[a] != b:
                return False
            if b in word_map and word_map[b] != a:
                return False
            pattern_map[a] = b
            word_map[b] = a
        return True
```
