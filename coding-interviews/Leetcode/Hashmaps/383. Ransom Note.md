from collections import defaultdict
class Solution:
    def canConstruct(self, ransomNote: str, magazine: str) -> bool:
        # input = ransomNote (string), magazine (string)
        # count all letters in magazine
        # for each letter in ransomnote take it away from magazine
        # if you have n one of that letter return false
        # if for every letter you can minus from magazine
        # return true
        # output = bool
        count = defaultdict(int)
        for l in magazine:
            count[l] +=1

        for n in ransomNote:
            if count[n] > 0: # we use n here because we are checking if there are any n's left in magazine
                count[n] -=1
            else:
                return False
        return True
